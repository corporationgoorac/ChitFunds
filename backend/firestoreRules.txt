rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // -------------------------------------------------------------------------
    // SECURITY ENGINE: WORKSPACE ADMIN VERIFICATION
    // This function guarantees that the request is coming from an authenticated 
    // user who holds a valid profile in the 'admins' collection with 'admin: true'.
    // -------------------------------------------------------------------------
    function isWorkspaceAdmin() {
      // Ensure the user is actually logged in via Firebase Auth
      let isAuthenticated = request.auth != null;
      
      // Point dynamically to the user's specific document in the admins collection
      let adminProfilePath = /databases/$(database)/documents/admins/$(request.auth.uid);
      
      // Validate that the profile exists AND the admin boolean is strictly true
      return isAuthenticated && 
             exists(adminProfilePath) && 
             get(adminProfilePath).data.admin == true;
    }

    // -------------------------------------------------------------------------
    // GLOBAL DATABASE ROUTING
    // Applies the logic across all collections (users, groups, transactions, etc.)
    // -------------------------------------------------------------------------
    match /{document=**} {
      
      // REQUIREMENT 1: Public Read Access
      // Anyone (even without logging in) can read the database.
      allow read: if true;
      
      // REQUIREMENT 2: Restricted Write Access
      // Only users who pass the isWorkspaceAdmin() function can create, edit, or delete data.
      allow write: if isWorkspaceAdmin();
      
    }
  }
}
