rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // -------------------------------------------------------------------------
    // SECURITY ENGINE: WORKSPACE ADMIN VERIFICATION
    // -------------------------------------------------------------------------
    function isWorkspaceAdmin() {
      let isAuthenticated = request.auth != null;
      let adminProfilePath = /databases/$(database)/documents/admins/$(request.auth.uid);
      
      return isAuthenticated && 
             exists(adminProfilePath) && 
             get(adminProfilePath).data.admin == true;
    }

    // -------------------------------------------------------------------------
    // 1. SPECIFIC ROUTING: ADMINS COLLECTION (Fixes Profile Setup)
    // -------------------------------------------------------------------------
    match /admins/{userId} {
      // Allows new users to complete their profile setup in login.html.
      // CRITICAL SECURITY: They can only save to their own UID, and the request 
      // is immediately rejected if they try to grant themselves 'admin: true'.
      allow write: if request.auth != null && 
                      request.auth.uid == userId && 
                      (!("admin" in request.resource.data) || request.resource.data.admin == false);
    }

    // -------------------------------------------------------------------------
    // 2. GLOBAL ROUTING: ALL COLLECTIONS (Users, Groups, Transactions, etc.)
    // Note: In Firestore, if ANY rule evaluates to true, access is granted.
    // -------------------------------------------------------------------------
    match /{document=**} {
      
      // REQUIREMENT 1: Public Read Access
      allow read: if true;
      
      // REQUIREMENT 2: Restricted Write Access
      // This allows fully verified Admins to write anywhere in the database, 
      // including updating other admins or promoting them to 'admin: true'.
      allow write: if isWorkspaceAdmin();
      
    }
  }
}
